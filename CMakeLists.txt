cmake_minimum_required(VERSION 3.28)
project(obs-auto-crop VERSION 0.1.0 LANGUAGES C CXX)

find_package(libobs REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Core Widgets)

# Ubuntu's libobs-dev may not ship a frontend API CMake package.
find_package(obs-frontend-api QUIET)
if(TARGET OBS::obs-frontend-api)
  set(obs_frontend_target OBS::obs-frontend-api)
elseif(TARGET OBS::frontend-api)
  set(obs_frontend_target OBS::frontend-api)
elseif(UNIX AND NOT APPLE)
  find_path(OBS_FRONTEND_INCLUDE_DIR obs-frontend-api.h PATH_SUFFIXES obs)
  find_library(OBS_FRONTEND_LIBRARY NAMES obs-frontend-api)
  if(NOT OBS_FRONTEND_INCLUDE_DIR OR NOT OBS_FRONTEND_LIBRARY)
    message(FATAL_ERROR "obs-frontend-api development files were not found")
  endif()
  add_library(obs-frontend-api-system UNKNOWN IMPORTED)
  set_target_properties(obs-frontend-api-system PROPERTIES
    IMPORTED_LOCATION "${OBS_FRONTEND_LIBRARY}"
    INTERFACE_INCLUDE_DIRECTORIES "${OBS_FRONTEND_INCLUDE_DIR}")
  set(obs_frontend_target obs-frontend-api-system)
else()
  message(FATAL_ERROR "obs-frontend-api CMake target was not found")
endif()

add_library(obs-auto-crop MODULE src/plugin-main.cpp src/black-border.cpp)
target_compile_features(obs-auto-crop PRIVATE cxx_std_17)
target_link_libraries(obs-auto-crop PRIVATE OBS::libobs ${obs_frontend_target} Qt6::Core Qt6::Widgets)
set_target_properties(obs-auto-crop PROPERTIES PREFIX "" OUTPUT_NAME "obs-auto-crop")

if(WIN32)
  install(TARGETS obs-auto-crop LIBRARY DESTINATION plugins/obs-auto-crop/bin/64bit RUNTIME DESTINATION plugins/obs-auto-crop/bin/64bit)
  install(DIRECTORY data/ DESTINATION plugins/obs-auto-crop/data)
elseif(APPLE)
  set_target_properties(obs-auto-crop PROPERTIES BUNDLE TRUE BUNDLE_EXTENSION plugin
    MACOSX_BUNDLE_GUI_IDENTIFIER "com.obsautocrop.plugin")
  file(GLOB_RECURSE locale_files CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/data/*")
  foreach(locale_file IN LISTS locale_files)
    file(RELATIVE_PATH locale_path "${CMAKE_CURRENT_SOURCE_DIR}/data" "${locale_file}")
    get_filename_component(locale_dir "${locale_path}" DIRECTORY)
    set_source_files_properties("${locale_file}" PROPERTIES MACOSX_PACKAGE_LOCATION "Resources/${locale_dir}")
  endforeach()
  target_sources(obs-auto-crop PRIVATE ${locale_files})
  install(TARGETS obs-auto-crop LIBRARY DESTINATION .)
else()
  include(GNUInstallDirs)
  install(TARGETS obs-auto-crop LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR}/obs-plugins)
  install(DIRECTORY data/ DESTINATION ${CMAKE_INSTALL_DATADIR}/obs/obs-plugins/obs-auto-crop)
endif()
